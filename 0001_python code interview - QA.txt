def find_duplicates(numbers):

    return list(duplicates)

import time


def retry_operation(operation, retries=3):

    for attempt in range(1, retries + 1):

        try:
            return operation()

        except Exception as e:

            print(f"Attempt {attempt} failed: {e}")

            if attempt < retries:
                time.sleep(2)

def flatten(data):

    result = []

    for item in data:

        if isinstance(item, list):
            result.extend(flatten(item))
        else:

    return result

 40. Class method vs static method


class Employee:

    company = "ABC"

    def __init__(self, name):
        self.name = name

    @classmethod
    def get_company(cls):
        return cls.company

    @staticmethod
    def add(a, b):
        return a + b


Usage:


print(Employee.get_company())

print(Employee.add(10, 20))


Interview answer:

`classmethod` receives `cls` and can access class-level data. `staticmethod` doesn't receive either `self` or `cls` and behaves like a normal utility function inside the class.

def all_pairs(numbers, target):

    seen = set()
    pairs = set()

    for number in numbers:

        complement = target - number

        if complement in seen:

 55. Find longest substring without repeating characters


def longest_unique_substring(text):

    seen = set()
    left = 0
    max_length = 0

    for right in range(len(text)):

        while text[right] in seen:
            seen.remove(text[left])
            left += 1

        seen.add(text[right])

        max_length = max(
            max_length,
            right - left + 1
        )

    return max_length


print(longest_unique_substring("abcabcbb"))


Interview concept: Sliding window.

 62. Retry with exponential backoff

This is a very good production interview scenario.


import time


def retry(operation, retries=3):

    for attempt in range(retries):

        try:
            return operation()

        except Exception:

            if attempt == retries - 1:
                raise

            delay = 2  attempt

            time.sleep(delay)


The delays are:


1 sec
2 sec
4 sec


Interview answer:

I use exponential backoff for transient failures such as network or external API failures. I also put a maximum retry limit to avoid continuously hammering the downstream service.

 64. Rate limiter

Scenario: Allow only 5 requests within a time window.


import time


class RateLimiter:

    def __init__(self, limit, window):
        self.limit = limit
        self.window = window
        self.requests = []

    def allow_request(self):

        current_time = time.time()

        self.requests = [
            request_time
            for request_time in self.requests
            if current_time - request_time < self.window
        ]

        if len(self.requests) >= self.limit:
            return False

        self.requests.append(current_time)

        return True


Production discussion:
For distributed applications, I would normally implement rate limiting using a shared store such as Redis rather than an in-memory list.

 66. Producer-consumer pattern


from queue import Queue
import threading
import time


queue = Queue()


def producer():

    for i in range(5):
        queue.put(i)
        print("Produced:", i)


def consumer():

    while True:

        item = queue.get()

        if item is None:
            break

        print("Consumed:", item)

        queue.task_done()


producer_thread = threading.Thread(
    target=producer
)

consumer_thread = threading.Thread(
    target=consumer
)

consumer_thread.start()
producer_thread.start()

producer_thread.join()

queue.put(None)

consumer_thread.join()


Interview concept: Thread-safe communication between producer and consumer.

 67. Thread-safe counter


import threading


class Counter:

    def __init__(self):
        self.value = 0
        self.lock = threading.Lock()

    def increment(self):

        with self.lock:
            self.value += 1


counter = Counter()

threads = [
    threading.Thread(
        target=counter.increment
    )
    for _ in range(100)
]

for thread in threads:
    thread.start()

for thread in threads:
    thread.join()

print(counter.value)


Interview answer:

I use a lock when multiple threads modify shared mutable state to prevent race conditions.

 71. Handle partial API failures


import asyncio


async def api_call(name):

    if name == "API-2":
        raise Exception("API failed")

    return f"{name} success"


async def main():

    results = await asyncio.gather(
        api_call("API-1"),
        api_call("API-2"),
        api_call("API-3"),
        return_exceptions=True
    )

    for result in results:

        if isinstance(result, Exception):
            print("Failed:", result)
        else:
            print(result)


asyncio.run(main())

 83. Memory-efficient processing with generator

        i for i in range(10_000_000)

def get_numbers():

def add_employee(

):

    return employees
